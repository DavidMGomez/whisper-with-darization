# openai-whisper's last PyPI release (20240930) is a source-only sdist whose
# legacy setup.py does `import pkg_resources`; that fails to build under
# uv/pip's isolated build env with a current setuptools ("ModuleNotFoundError:
# No module named 'pkg_resources'") -- a pre-existing packaging break
# unrelated to this migration, hit as soon as anyone tries a fresh build.
# Installing from GitHub HEAD (same pattern already used for whisperx below)
# picks up any packaging fixes made since that last PyPI release.
setuptools<81
ffmpeg-python

# Pinned for the same reason as nemo_toolkit was: unpinned speechbrain let
# pip backtrack all the way down to the 0.5.x line (years old), which pins
# torch<=1.11, conflicting with our torch==2.4 pin.
speechbrain==1.1.0
# nvidia/Nemotron-3-Diarization's own README says to install transformers

# classes aren't in any PyPI transformers release yet. Without this,
# AutoProcessor.from_pretrained fails with "Unrecognized processing class".

# librosa for audio feature extraction ("NemotronAsrStreamingFeatureExtractor
# requires the librosa library").
librosa
